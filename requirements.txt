# md_to_pdf.py uses only the Python standard library; there is nothing to
# pip install for the default (xelatex) setup.
#
# Optional: weasyprint as an alternative PDF engine, used only when xelatex
# and pdflatex are not installed. Uncomment to install it with
# `pip install -r requirements.txt`.
# weasyprint>=66.0
#
# External system dependencies (not installed via pip):
# - pandoc:        brew install pandoc  /  sudo apt-get install pandoc
# - xelatex:       MacTeX  /  sudo apt-get install texlive-xetex
# - Japanese text: xeCJK and the Harano Aji fonts (in MacTeX and texlive-full;
#                  on Debian/Ubuntu: texlive-lang-cjk texlive-lang-chinese
#                  texlive-lang-japanese — xeCJK needs ctexhook.sty from
#                  the Chinese package)
# - Node.js 22.13+ (mermaid-cli is installed with the npm package)
